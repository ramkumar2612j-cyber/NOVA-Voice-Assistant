import com.nova.assistant.Fbank
import java.io.File
fun main(a: Array<String>) {
    val pcm = File(a[0]).readBytes()
    val t0 = System.nanoTime()
    val f = Fbank.compute(Fbank.pcmToSamples(pcm))!!
    val ms = (System.nanoTime() - t0) / 1e6
    File(a[1]).writeText(f.joinToString("\n") { it.toString() })
    println("frames=" + f.size / Fbank.MELS + " ms=" + ms)
}
